-- Roblox AI Developer
-- Local plugin prototype

local toolbar = plugin:CreateToolbar("Roblox AI Developer")

local openButton = toolbar:CreateButton(
    "OpenAI",
    "Open Roblox AI Developer",
    ""
)

openButton.ClickableWhenViewportHidden = true

local widgetInfo = DockWidgetPluginGuiInfo.new(
    Enum.InitialDockState.Right,
    false,
    false,
    420,
    600,
    320,
    400
)

local widget = plugin:CreateDockWidgetPluginGui(
    "RobloxAIDeveloperWidget",
    widgetInfo
)

widget.Title = "Roblox AI Developer"

local frame = Instance.new("Frame")
frame.Size = UDim2.fromScale(1, 1)
frame.BackgroundColor3 = Color3.fromRGB(30, 32, 40)
frame.Parent = widget

local title = Instance.new("TextLabel")
title.Size = UDim2.new(1, -20, 0, 48)
title.Position = UDim2.fromOffset(10, 8)
title.BackgroundTransparency = 1
title.Text = "Roblox AI Developer"
title.TextColor3 = Color3.new(1, 1, 1)
title.TextSize = 22
title.Font = Enum.Font.GothamBold
title.Parent = frame

local description = Instance.new("TextLabel")
description.Size = UDim2.new(1, -20, 0, 50)
description.Position = UDim2.fromOffset(10, 58)
description.BackgroundTransparency = 1
description.Text = "Describe what you want to build in your game."
description.TextColor3 = Color3.fromRGB(200, 205, 220)
description.TextSize = 14
description.TextWrapped = true
description.Parent = frame

local prompt = Instance.new("TextBox")
prompt.Size = UDim2.new(1, -20, 0, 130)
prompt.Position = UDim2.fromOffset(10, 118)
prompt.BackgroundColor3 = Color3.fromRGB(45, 48, 60)
prompt.TextColor3 = Color3.new(1, 1, 1)
prompt.PlaceholderText = "Example: Create a speed upgrade shop..."
prompt.Text = ""
prompt.TextSize = 14
prompt.TextWrapped = true
prompt.MultiLine = true
prompt.ClearTextOnFocus = false
prompt.Parent = frame

local proposal = Instance.new("TextLabel")
proposal.Size = UDim2.new(1, -20, 0, 180)
proposal.Position = UDim2.fromOffset(10, 258)
proposal.BackgroundColor3 = Color3.fromRGB(40, 43, 53)
proposal.TextColor3 = Color3.fromRGB(220, 225, 235)
proposal.TextSize = 14
proposal.TextWrapped = true
proposal.TextYAlignment = Enum.TextYAlignment.Top
proposal.TextXAlignment = Enum.TextXAlignment.Left
proposal.Text = "Your proposed changes will appear here."
proposal.Parent = frame

local prepareButton = Instance.new("TextButton")
prepareButton.Size = UDim2.new(1, -20, 0, 42)
prepareButton.Position = UDim2.fromOffset(10, 448)
prepareButton.BackgroundColor3 = Color3.fromRGB(65, 105, 210)
prepareButton.TextColor3 = Color3.new(1, 1, 1)
prepareButton.Text = "Prepare Proposal"
prepareButton.TextSize = 16
prepareButton.Parent = frame

local approveButton = Instance.new("TextButton")
approveButton.Size = UDim2.new(1, -20, 0, 42)
approveButton.Position = UDim2.fromOffset(10, 498)
approveButton.BackgroundColor3 = Color3.fromRGB(45, 145, 90)
approveButton.TextColor3 = Color3.new(1, 1, 1)
approveButton.Text = "Approve & Apply"
approveButton.TextSize = 16
approveButton.Parent = frame

local pendingPrompt = nil

prepareButton.MouseButton1Click:Connect(function()
    pendingPrompt = nil

    local request = prompt.Text:match("^%s*(.-)%s*$")

    if request == "" then
        proposal.Text = "Please describe what you want to build first."
        return
    end

    pendingPrompt = request

    proposal.Text =
        "PROPOSAL PREVIEW\n\n" ..
        "Request: " .. request ..
        "\n\nThis prototype does not make any changes yet." ..
        "\n\nNext, we will connect the AI so it can plan real Roblox changes."
end)

approveButton.MouseButton1Click:Connect(function()
    if not pendingPrompt then
        proposal.Text = "Prepare a proposal first. Nothing has been changed."
        return
    end

    proposal.Text =
        "Approval received for:\n\n" .. pendingPrompt ..
        "\n\nNo changes were made. The AI execution system is not connected yet."
end)

openButton.Click:Connect(function()
    widget.Enabled = not widget.Enabled
end)

print("Roblox AI Developer plugin loaded.")
